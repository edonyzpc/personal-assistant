/* Read/fixture-only HEIC host investigation; no plugin feature or decoder dependency. */
globalThis.runB129HeicCoreCheck = async (app, runId) => {
  if (app.vault.getName() !== 'test' || !/^[a-zA-Z0-9_-]+$/.test(runId)) throw new Error('test_scope_required');
  const root = `b129-heic-core-check/${runId}`;
  if (await app.vault.adapter.exists(root)) throw new Error('unique_run_required');
  await app.vault.createFolder(root);
  const hash = async (bytes) => Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256', bytes)))
    .map(b => b.toString(16).padStart(2, '0')).join('');
  const files = ['b129-p0-fixtures/chart-sips.heic', 'b129-p0-fixtures/chart-exif-orientation-6.jpg'];
  const result = { kind: 'b129-heic-core-check', runId, createdAt: new Date().toISOString(),
    userAgent: navigator.userAgent, registry: {}, native: [], markdown: [], conversions: [], errors: [] };
  const mimeByExt = { heic: 'image/heic', jpg: 'image/jpeg' };
  for (const ext of ['heic', 'heif', 'jpg', 'png']) {
    result.registry[ext] = { view: app.viewRegistry.getTypeByExtension(ext) ?? null,
      embed: app.embedRegistry.isExtensionRegistered(ext) };
  }
  const load = async (src) => {
    const img = new Image(); let timer;
    try {
      await new Promise((resolve,reject) => {
        timer = setTimeout(() => reject(new Error('image_timeout')), 8000);
        img.onload = resolve; img.onerror = () => reject(new Error('image_error')); img.src = src;
      });
      return img;
    } catch (e) { img.src = ''; throw e; }
    finally { clearTimeout(timer); }
  };
  for (const path of files) {
    const file = app.vault.getAbstractFileByPath(path);
    if (!file) throw new Error(`missing_fixture:${path}`);
    const bytes = await app.vault.readBinary(file);
    const originalHash = await hash(bytes);
    const resource = app.vault.getResourcePath(file);
    for (const sourceKind of ['vault-resource', 'blob']) {
      const src = sourceKind === 'blob' ? URL.createObjectURL(new Blob([bytes], {type:mimeByExt[file.extension]})) : resource;
      let img;
      try {
        img = await load(src);
        const entry = { path, originalHash, sourceKind, decoded: true, width: img.naturalWidth, height: img.naturalHeight };
        result.native.push(entry);
        if (file.extension === 'heic' && sourceKind === 'blob') {
          const canvas = document.createElement('canvas');
          try {
            canvas.width = img.naturalWidth; canvas.height = img.naturalHeight;
            const context = canvas.getContext('2d');
            context.fillStyle = '#ffffff'; context.fillRect(0,0,canvas.width,canvas.height);
            context.drawImage(img,0,0);
            let timer;
            const jpeg = await new Promise((resolve,reject) => {
              timer=setTimeout(()=>reject(new Error('encode_timeout')),8000);
              canvas.toBlob(b=>b?resolve(b):reject(new Error('encode_failed')),'image/jpeg',0.9);
            }).finally(()=>clearTimeout(timer));
            const jpegBytes = await jpeg.arrayBuffer();
            const output = `${root}/converted-native.jpg`;
            await app.vault.createBinary(output,jpegBytes);
            result.conversions.push({ path, output, bytes:jpegBytes.byteLength, type:jpeg.type,
              sha256:await hash(jpegBytes), originalUnchanged:originalHash===await hash(await app.vault.readBinary(file)) });
          } finally { canvas.width=0;canvas.height=0; }
        }
      } catch(e) { result.native.push({path,originalHash,sourceKind,decoded:false,error:String(e.message)}); }
      finally { if(img)img.src='';if(sourceKind==='blob')URL.revokeObjectURL(src); }
    }
  }
  const notePath = `${root}/core-render.md`;
  let noteText = '# B-129 HEIC core comparison\n\n';
  const cases = [
    {name:'HEIC wiki embed',md:`![[${files[0]}]]`},
    {name:'HEIC Markdown image',md:`![](${files[0]})`},
    {name:'JPEG control',md:`![[${files[1]}]]`},
  ];
  for (const converted of result.conversions) cases.push({name:'Native HEIC to JPEG',md:`![[${converted.output}]]`});
  for(const item of cases) noteText += `## ${item.name}\n\n${item.md}\n\n`;
  await app.vault.create(notePath,noteText);
  result.notePath=notePath;
  const leaf=app.workspace.getLeaf(false);
  await leaf.setViewState({type:'markdown',state:{file:notePath,mode:'preview'},active:true});
  await new Promise(resolve=>setTimeout(resolve,500));
  const container=leaf.view.contentEl;
  result.markdown.push({text:container.textContent,
    embeds:Array.from(container.querySelectorAll('.internal-embed')).map(e=>({src:e.getAttribute('src'),classes:e.className,text:e.textContent})),
    images:Array.from(container.querySelectorAll('img')).map(img=>({src:img.getAttribute('src'),complete:img.complete,width:img.naturalWidth,height:img.naturalHeight}))});
  result.finishedAt=new Date().toISOString();
  await app.vault.adapter.write(`${root}/receipt.json`,JSON.stringify(result,null,2));
  return result;
};
