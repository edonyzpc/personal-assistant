/* Targeted macOS-host conversion experiment; no production or dependency changes. */
globalThis.runB129MacHeicConversion = async (app, variant = 'plain') => {
  if (app.vault.getName() !== 'test' || process.platform !== 'darwin') throw new Error('mac_test_only');
  if (!['plain','gps'].includes(variant)) throw new Error('fixture_required');
  const fs = require('node:fs/promises');
  const path = require('node:path');
  const cp = require('node:child_process');
  const dir = await fs.mkdtemp(path.join(require('node:os').tmpdir(), 'b129-heic-convert-'));
  const pngPath = path.join(dir, 'intermediate.png');
  const source = variant === 'gps' ? 'b129-heic-core-check/chart-synthetic-gps.heic' : 'b129-p0-fixtures/chart-sips.heic';
  const output = variant === 'gps' ? 'b129-heic-core-check/mac-native-gps-converted.jpg' : 'b129-heic-core-check/mac-native-converted.jpg';
  if (await app.vault.adapter.exists(output)) throw new Error('output_exists');
  const sha = data => require('node:crypto').createHash('sha256').update(new Uint8Array(data)).digest('hex');
  const before = sha(await app.vault.adapter.readBinary(source));
  const started = performance.now();
  const result = { kind: 'b129-macos-heic-conversion', source, sourceSha256: before, output, createdAt: new Date().toISOString() };
  let url; const img = new Image(); const canvas = document.createElement('canvas');
  try {
    const native = await new Promise((resolve, reject) => cp.execFile('/usr/bin/sips',
      ['-s', 'format', 'png', path.join(app.vault.adapter.basePath, source), '--out', pngPath],
      { timeout: 15000, maxBuffer: 8192 }, (error, stdout, stderr) => error ? reject(error) : resolve({stdout,stderr})));
    result.stderr = native.stderr;
    const bytes = await fs.readFile(pngPath);
    result.intermediateBytes = bytes.length;
    result.intermediateSha256 = sha(bytes);
    url = URL.createObjectURL(new Blob([bytes], {type:'image/png'}));
    let timer;
    await new Promise((resolve,reject)=>{
      timer=setTimeout(()=>reject(new Error('decode_timeout')),8000);
      img.onload=resolve;img.onerror=()=>reject(new Error('png_decode_failed'));img.src=url;
    }).finally(()=>clearTimeout(timer));
    canvas.width=img.naturalWidth;canvas.height=img.naturalHeight;
    const context=canvas.getContext('2d');context.fillStyle='#ffffff';context.fillRect(0,0,canvas.width,canvas.height);
    context.drawImage(img,0,0);
    result.width=canvas.width;result.height=canvas.height;
    result.corners=[[.05,.05],[.95,.05],[.05,.95],[.95,.95]].map(([x,y])=>[...context.getImageData(Math.floor(x*canvas.width),Math.floor(y*canvas.height),1,1).data]);
    let encodeTimer;
    const jpeg=await new Promise((resolve,reject)=>{
      encodeTimer=setTimeout(()=>reject(new Error('encode_timeout')),8000);
      canvas.toBlob(b=>b?resolve(b):reject(new Error('encode_failed')),'image/jpeg',0.9);
    }).finally(()=>clearTimeout(encodeTimer));
    const jpegBytes=await jpeg.arrayBuffer();
    await app.vault.createBinary(output,jpegBytes);
    result.outputBytes=jpegBytes.byteLength;result.outputSha256=sha(jpegBytes);result.outputMime=jpeg.type;
    result.sourceUnchanged=before===sha(await app.vault.adapter.readBinary(source));
    result.elapsedMs=performance.now()-started;
  } finally { img.src='';if(url)URL.revokeObjectURL(url);canvas.width=0;canvas.height=0;
    // Delete only the exact temporary artifact generated by this run, outside the vault.
    await fs.unlink(pngPath).catch(()=>{});await fs.rmdir(dir).catch(()=>{}); }
  await app.vault.adapter.write(output.replace('.jpg','-receipt.json'),JSON.stringify(result,null,2));
  return result;
};
